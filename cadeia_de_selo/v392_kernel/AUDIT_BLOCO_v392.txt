-- KH-N1 (v392; desenho (a), decisao da gerencia c1 de 13/10/2026): o axioma de TODA constante dos modulos TGL e TGLExt, no formato do #print axioms, mais a linha TGL_AXIOM_COVERAGE_TOTAL; fonte = F_KERNEL/kh/KH_formato_umpy_total.lean (sha256 cb0523546bad415d4b696ee7c7ed41c6ff36c8bc9d16357a155d63a15325065b)
open Lean Elab Command in
run_cmd do
  let env ← getEnv
  let mut n_decl := 0
  let mut n_thm := 0
  let mut n_fora := 0
  for (n, ci) in env.constants.map₁.toList do
    match env.getModuleIdxFor? n with
    | none => pure ()
    | some idx =>
      let m := env.header.moduleNames[idx.toNat]!
      if m.getRoot == `TGL || m.getRoot == `TGLExt then
        let axs ← liftCoreM (collectAxioms n)
        n_decl := n_decl + 1
        if ci matches .thmInfo _ then n_thm := n_thm + 1
        unless axs.all (fun a => a == `propext || a == `Classical.choice || a == `Quot.sound) do n_fora := n_fora + 1
        if axs.isEmpty then
          IO.println ("'" ++ n.toString ++ "' does not depend on any axioms")
        else
          IO.println ("'" ++ n.toString ++ "' depends on axioms: [" ++ String.intercalate ", " (axs.toList.map Name.toString) ++ "]")
  IO.println ("TGL_AXIOM_COVERAGE_TOTAL decl=" ++ toString n_decl ++ " thm=" ++ toString n_thm ++ " fora_do_trio=" ++ toString n_fora)
